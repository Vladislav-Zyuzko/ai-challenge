import { readFile } from 'node:fs/promises';

async function readJson(path) {
  let text;
  try {
    text = await readFile(path, 'utf8');
  } catch (err) {
    if (err.code === 'ENOENT') throw new Error(`File not found: ${path}`, { cause: err });
    if (err.code === 'EACCES') throw new Error(`Permission denied: ${path}`, { cause: err });
    throw err;
  }

  try {
    return JSON.parse(text);
  } catch (err) {
    throw new SyntaxError(`Invalid JSON in ${path}: ${err.message}`, { cause: err });
  }
}

try {
  const data = await readJson('./config.json');
  console.log(data);
} catch (err) {
  console.error(err.message);
  process.exitCode = 1;
}
