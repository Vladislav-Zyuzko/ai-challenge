```js
const fs = require('node:fs/promises');

async function readJson(path) {
  let raw;
  try {
    raw = await fs.readFile(path, 'utf8');
  } catch (err) {
    if (err.code === 'ENOENT') throw new Error(`Файл не найден: ${path}`);
    if (err.code === 'EACCES') throw new Error(`Нет доступа: ${path}`);
    throw err;                       // прочие I/O-ошибки наверх
  }
  try {
    return JSON.parse(raw);
  } catch (err) {
    throw new Error(`Некорректный JSON в ${path}: ${err.message}`);
  }
}
```

Синхронный вариант: `JSON.parse(fs.readFileSync(path, 'utf8'))` в одном `try/catch`, но различать `err.code === 'ENOENT'` и `err instanceof SyntaxError`. `require('./data.json')` кэширует результат и падает процессом при ошибке — для динамического чтения не годится.
