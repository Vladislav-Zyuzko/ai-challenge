```js
import { readFile } from 'node:fs/promises';

try {
  const data = JSON.parse(await readFile('./config.json', 'utf8'));
  console.log(data);
} catch (err) {
  if (err.code === 'ENOENT') console.error('файл не найден:', err.path);
  else if (err instanceof SyntaxError) console.error('невалидный JSON:', err.message);
  else throw err;
}
```
